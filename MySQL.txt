SHOW DATABASES;

CREATE DATABASE bookstore;

USE bookstore;

CREATE TABLE books (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(50),
    author VARCHAR(50),
    price INT
);
